import logging
from logging.handlers import RotatingFileHandler

handler = RotatingFileHandler(
    "/var/log/myapp.log",
    maxBytes=10*1024*1024,
    backupCount=5
)

logging.basicConfig(level=logging.WARNING, handlers=[handler])




# 1. backup
sudo cp /etc/systemd/resolved.conf /etc/systemd/resolved.conf.bak.$(date +%F)

# 2. adauga serve-stale + cache sub [Resolve]

EOF
